def run_monte_carlo_simulation(raw_holdings: str, num_simulations: int = 10000, 
                                days_ahead: int = 252, portfolio_value: float = None) -> dict:
    """
    Run a Monte Carlo simulation projecting portfolio value into the future
    based on historical return distributions and the Cholesky decomposition 
    of the correlation matrix.
    """
    holdings = parse_holdings(raw_holdings)
    if not holdings:
        return {}

    symbols = [h["symbol"] for h in holdings]
    prices  = _fetch_prices(symbols, period="2y").drop(columns=[BENCHMARK], errors="ignore")
    if prices.empty:
        return {}

    total_val, w_dict = _get_weights(holdings, prices, portfolio_value)
    if total_val == 0.0:
        return {}

    weights = np.array([w_dict.get(s, 0.0) for s in prices.columns])
    
    # Calculate daily returns, mean returns, and covariance matrix
    daily_returns = prices.pct_change().dropna()
    mean_returns = daily_returns.mean().values
    cov_matrix = daily_returns.cov().values

    # Perform Cholesky decomposition
    try:
        L = np.linalg.cholesky(cov_matrix)
    except np.linalg.LinAlgError:
        # Fallback if matrix is not positive-definite
        return {"error": "Covariance matrix is not positive-definite. Cannot run Monte Carlo."}

    # Simulation setup
    num_assets = len(mean_returns)
    simulation_res = np.zeros((days_ahead, num_simulations))
    
    # Run simulations
    for i in range(num_simulations):
        # Generate independent standard normal variables
        Z = np.random.normal(0, 1, (days_ahead, num_assets))
        # Correlate them
        correlated_returns = Z.dot(L.T)
        # Add drift (mean)
        daily_sim_returns = correlated_returns + mean_returns
        
        # Calculate daily portfolio returns for this simulation path
        port_sim_returns = daily_sim_returns.dot(weights)
        
        # Calculate cumulative returns and apply to initial portfolio value
        cumulative_returns = np.cumprod(1 + port_sim_returns)
        simulation_res[:, i] = total_val * cumulative_returns

    # Extract final values across all simulations
    final_values = simulation_res[-1, :]
    
    return {
        "mean_expected_value": float(np.mean(final_values)),
        "median_expected_value": float(np.median(final_values)),
        "worst_1_percentile": float(np.percentile(final_values, 1)),
        "worst_5_percentile": float(np.percentile(final_values, 5)),
        "best_5_percentile": float(np.percentile(final_values, 95)),
        "portfolio_value": total_val,
        "days_projected": days_ahead,
        "num_simulations": num_simulations
    }
